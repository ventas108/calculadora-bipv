    20  def obtener_constantes_tecnologia(tecnologia: str) -> dict:
    21      """

    34  def calcular_rsh_cdte(G, R_sh_ref, c_Rsh=5.5, R_sh_base=0.0, G_ref=1000.0):
    35      """
    36      Rsh exponencial CdTe — Mermoud 2005.
    37      Equivalente al bloque interno de TrasladarParametrosGT (VBA).
    38
    39      Rsh(G) = R_sh_ref × exp(−c_Rsh × (G/G_ref − 1)) + R_sh_base
    40      """
    41      G_arr  = np.atleast_1d(np.asarray(G, dtype=float))
    42      G_safe = np.where(G_arr > 0, G_arr, 1.0)
    43      rsh    = R_sh_ref * np.exp(-c_Rsh * (G_safe / G_ref - 1.0)) + R_sh_base
    44      return float(rsh) if rsh.size == 1 else rsh

    47  def trasladar_parametros_gt(G, T_cel_C, panel: dict):
    48      """
    49      Equivalente Python de TrasladarParametrosGT (VBA, Mod_ModeloDiodo).
    50      Traduce los parámetros SDM de STC a condiciones reales (G, T_celda).
    51
    52      Usa pvlib.calcparams_desoto() para I_L, I_o, nNsVth, R_s
    53      y reemplaza R_sh por el modelo exponencial CdTe de Mermoud 2005.
    54      """
    55      constantes = obtener_constantes_tecnologia(panel["tecnologia"])
    56
    57      # nNsVth_ref en Voltios (pvlib espera Voltios, no adimensional)
    58      Vt_ref      = K_BOLTZMANN * T_REF_K / Q_ELECTRON   # 0.025693 V @ 25°C
    59      nNsVth_ref  = panel["a_ref"] * Vt_ref               # 154 × 0.025693 = 3.957 V
    60
    61      # R_sh exponencial CdTe (Mermoud 2005) — reemplaza el lineal de pvlib
    62      R_sh_exp = calcular_rsh_cdte(
    63          G,
    64          panel["R_sh_ref"],
    65          c_Rsh     = constantes["c_Rsh"],
    66          R_sh_base = panel.get("R_sh_base", 0.0),
    67      )
    68
    69      # pvlib.calcparams_desoto para I_L, I_o, R_s, nNsVth
    70      I_L, I_o, R_s, _R_sh_pvlib, nNsVth = pvlib.pvsystem.calcparams_desoto(
    71          effective_irradiance = G,
    72          temp_cell            = T_cel_C,
    73          alpha_sc             = panel["Tk_alfa"] / 100.0,
    74          a_ref                = nNsVth_ref,
    75          I_L_ref              = panel["I_L_ref"],
    76          I_o_ref              = panel["I_o_ref"],
    77          R_sh_ref             = panel["R_sh_ref"],
    78          R_s                  = panel["R_s"],
    79          EgRef                = constantes["Eg_ref"],
    80          dEgdT                = constantes["dEgdT"],
    81          irrad_ref            = G_REF,
    82          temp_ref             = 25.0,
    83      )
    84
    85      # Devuelve con Rsh exponencial CdTe en lugar del lineal de pvlib
    86      return I_L, I_o, R_s, R_sh_exp, nNsVth

    89  def resolver_curva_iv(G, T_cel_C, panel: dict, n_puntos=100):
    90      """
    91      Equivalente Python de CurvaIV_CdTe (VBA, SimuladorIV_CdTe_v2).
    92      Retorna curva I-V completa + puntos clave (Voc, Isc, Vmp, Imp, Pmax, FF).
    93      """
    94      if G <= 0:
    95          return {"Voc": 0, "Isc": 0, "Vmp": 0, "Imp": 0, "Pmax": 0, "FF": 0,
    96                  "V": None, "I": None}
    97
    98      I_L, I_o, R_s, R_sh, nNsVth = trasladar_parametros_gt(G, T_cel_C, panel)
    99
   100      resultado = pvlib.pvsystem.singlediode(
   101          photocurrent       = I_L,
   102          saturation_current = I_o,
   103          resistance_series  = R_s,
   104          resistance_shunt   = R_sh,
   105          nNsVth             = nNsVth,
   106          ivcurve_pnts       = n_puntos,
   107          method             = 'lambertw',
   108      )
   109
   110      Voc  = float(resultado['v_oc'])
   111      Isc  = float(resultado['i_sc'])
   112      Pmax = float(resultado['p_mp'])
   113      FF   = Pmax / (Voc * Isc) if (Voc * Isc) > 0 else 0.0
   114
   115      return {
   116          "Voc":  Voc,
   117          "Isc":  Isc,
   118          "Vmp":  float(resultado['v_mp']),
   119          "Imp":  float(resultado['i_mp']),
   120          "Pmax": Pmax,
   121          "FF":   FF,
   122          "V":    resultado.get('v', None),
   123          "I":    resultado.get('i', None),
   124      }

   141      """
   142      Compara el SDM calibrado contra los valores STC de la ficha.
